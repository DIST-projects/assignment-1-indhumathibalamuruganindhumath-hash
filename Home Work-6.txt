1) Simulate the undo and redo option in a word processor using a doubly linked list.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Node {
    char action[50];
    struct Node *prev, *next;
};

struct Node *current = NULL;

void addAction(char action[]) {
    struct Node *newNode = malloc(sizeof(struct Node));
    strcpy(newNode->action, action);
    newNode->prev = current;
    newNode->next = NULL;

    if (current != NULL) {
        struct Node *p = current->next;
        while (p != NULL) {
            struct Node *temp = p;
            p = p->next;
            free(temp);
        }
        current->next = newNode;
    }

    current = newNode;
}

void undo() {
    if (current == NULL)
        printf("Nothing to undo\n");
    else {
        printf("Undo: %s\n", current->action);
        current = current->prev;
    }
}

void redo() {
    if (current == NULL)
        printf("Nothing to redo\n");
    else if (current->next != NULL) {
        current = current->next;
        printf("Redo: %s\n", current->action);
    } else
        printf("Nothing to redo\n");
}

int main() {
    addAction("Type Hello");
    addAction("Type World");

    undo();
    redo();

    return 0;
}

Output:

Undo: Type World
Redo: Type World

3)  Swap two elements by adjusting only the pointers (and not the data) using a doubly linked list.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev, *next;
};

void swapAdjacent(struct Node **head,
                  struct Node *a,
                  struct Node *b) {
    if (a == NULL || b == NULL || a->next != b)
        return;

    struct Node *p = a->prev;
    struct Node *n = b->next;

    if (p != NULL)
        p->next = b;
    else
        *head = b;

    b->prev = p;
    b->next = a;
    a->prev = b;
    a->next = n;

    if (n != NULL)
        n->prev = a;
}

void display(struct Node *head) {
    while (head != NULL) {
        printf("%d ", head->data);
        head = head->next;
    }
    printf("\n");
}

int main() {
    struct Node n1 = {10, NULL, NULL};
    struct Node n2 = {20, NULL, NULL};
    struct Node n3 = {30, NULL, NULL};

    struct Node *head = &n1;

    n1.next = &n2;
    n2.prev = &n1;
    n2.next = &n3;
    n3.prev = &n2;
    printf("Before swapping: ");
    display(head);

    swapAdjacent(&head, &n1, &n2);
    printf("After swapping: ");
    display(head);

    return 0;
}

void deleteMiddle(struct Node **head) {
    if (*head == NULL)
        return;

    struct Node *slow = *head;
    struct Node *fast = *head;

    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
    }

    if (slow->prev != NULL)
        slow->prev->next = slow->next;
    else
        *head = slow->next;

    if (slow->next != NULL)
        slow->next->prev = slow->prev;

    free(slow);
}

8) Calculate the frequency count of a given integer in a circular doubly linked list.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev, *next;
};

int frequency(struct Node *head, int key) {
    if (head == NULL)
        return 0;

    int count = 0;
    struct Node *p = head;

    do {
        if (p->data == key)
            count++;

        p = p->next;
    } while (p != head);

    return count;
}

    struct Node n2 = {20, NULL, NULL};

    n3.next = &n4; n4.prev = &n3;
    n4.next = &n5; n5.prev = &n4;

    n5.next = head;
    head->prev = &n5;

10) Calculate the height and depth of a tree and print them.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->data = x;
    n->left = n->right = NULL;
    return n;
}

int height(struct Node *root) {
    if (root == NULL)
        return -1;

    int l = height(root->left);
    int r = height(root->right);

    return 1 + (l > r ? l : r);
}

int findDepth(struct Node *root, int key, int d) {
    if (root == NULL)
        return -1;

    if (root->data == key)
        return d;

    int result = findDepth(root->left, key, d + 1);

    if (result != -1)
        return result;

    return findDepth(root->right, key, d + 1);
}

int main() {
    struct Node *root = createNode(10);
    root->left = createNode(20);
    root->right = createNode(30);
    root->left->left = createNode(40);

    printf("Height of tree = %d\n", height(root));
    printf("Depth of node 40 = %d\n",
           findDepth(root, 40, 0));

    return 0;
}

12) Given a binary tree, connect all nodes at the same level using next-right pointers.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right, *nextRight;
};

void connectLevels(struct Node *root) {
    if (root == NULL)
        return;

    struct Node *queue[100];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        int count = rear - front;
        struct Node *prev = NULL;

        while (count--) {
            struct Node *cur = queue[front++];

            if (prev != NULL)
                prev->nextRight = cur;

            prev = cur;

            if (cur->left != NULL)
                queue[rear++] = cur->left;

            if (cur->right != NULL)
                queue[rear++] = cur->right;
        }

        prev->nextRight = NULL;
    }
}

13)  Implement a priority queue using a binary search tree with insertion, searching and deletion.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        root = malloc(sizeof(struct Node));
        root->data = x;
        root->left = root->right = NULL;
    } else if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

struct Node *search(struct Node *root, int x) {
    if (root == NULL || root->data == x)
        return root;

    if (x < root->data)
        return search(root->left, x);

    return search(root->right, x);
}

struct Node *deleteNode(struct Node *root, int x) {
    if (root == NULL)
        return NULL;

    if (x < root->data)
        root->left = deleteNode(root->left, x);
    else if (x > root->data)
        root->right = deleteNode(root->right, x);
    else {
        if (root->left == NULL) {
            struct Node *temp = root->right;
            free(root);
            return temp;
        }

        if (root->right == NULL) {
            struct Node *temp = root->left;
            free(root);
            return temp;
        }

        struct Node *temp = root->right;

        while (temp->left != NULL)
            temp = temp->left;

        root->data = temp->data;
        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

int getMin(struct Node *root) {
    while (root->left != NULL)
        root = root->left;

    return root->data;
}

int main() {
    struct Node *root = NULL;

    root = insert(root, 40);
    root = insert(root, 20);
    root = insert(root, 60);
    root = insert(root, 10);
    root = insert(root, 30);

    printf("Highest priority element = %d\n",
           getMin(root));

    if (search(root, 30) != NULL)
        printf("Element found\n");
    else
        printf("Element not found\n");

    root = deleteNode(root, 10);

    printf("After deletion, highest priority = %d\n",

    return 0;
}

15) Convert the given postfix expression into an expression tree.

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

struct Node {
    char data;
    struct Node *left, *right;
};

struct Node *createNode(char c) {
    struct Node *n = malloc(sizeof(struct Node));
    n->data = c;
    n->left = n->right = NULL;
    return n;
}

struct Node *buildTree(char exp[]) {
    struct Node *stack[100];
    int top = -1;

    for (int i = 0; exp[i] != '\0'; i++) {
        char c = exp[i];

        if (c == ' ')
            continue;

        struct Node *n = createNode(c);

        if (isalnum((unsigned char)c)) {
            stack[++top] = n;
        } else {
            n->right = stack[top--];
            n->left = stack[top--];
            stack[++top] = n;
        }
    }

    return top == 0 ? stack[top] : NULL;
}

void inorder(struct Node *root) {
    if (root == NULL)
        return;

    if (root->left != NULL || root->right != NULL)
        printf("(");

    inorder(root->left);
    printf("%c", root->data);
    inorder(root->right);

    if (root->left != NULL || root->right != NULL)
        printf(")");
}

int main() {
    struct Node *root = buildTree("AB+C*");

    printf("Infix expression: ");
    inorder(root);

    return 0;
}

17) Heap Sort for a Set of Numbers.

#include <stdio.h>

void heapify(int a[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;
    int temp;

    if (left < n && a[left] > a[largest])
        largest = left;

    if (right < n && a[right] > a[largest])
        largest = right;

    if (largest != i) {
        temp = a[i];
        a[i] = a[largest];
        a[largest] = temp;

        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n) {
    int i, temp;

    for (i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    for (i = n - 1; i > 0; i--) {
        temp = a[0];
        a[0] = a[i];
        a[i] = temp;

        heapify(a, i, 0);
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    heapSort(a, n);

    printf("Sorted elements: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n");
    return 0;
}
